// ? 閮擃?IP ?????(Rate Limiter for public write endpoints)
interface RateLimitBucket {
  count: number;
  resetAt: number;
}
const rateLimitStore = new Map<string, RateLimitBucket>();

export function createRateLimiter(maxRequests: number, windowMs: number = 60 * 1000, actionName: string = '??') {
  return (req: express.Request, res: express.Response, next: express.NextFunction) => {
    const ip = (req.headers['x-forwarded-for'] as string)?.split(',')[0]?.trim() || req.ip || '127.0.0.1';
    const key = `${actionName}:${ip}`;
    const now = Date.now();

    let bucket = rateLimitStore.get(key);
    if (!bucket || now > bucket.resetAt) {
      bucket = { count: 1, resetAt: now + windowMs };
      rateLimitStore.set(key, bucket);
      return next();
    }

    if (bucket.count >= maxRequests) {
      const waitSec = Math.ceil((bucket.resetAt - now) / 1000);
      return res.status(429).json({
        error: `隢??餌???嚗?{actionName} ?餌?撌脤?銝?嚗???${waitSec} 蝘??岫 (Too Many Requests)`
      });
    }

    bucket.count++;
    return next();
  };
}
